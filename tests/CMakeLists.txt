function(cr_add_test name)
    qt_add_executable(${name} ${name}.cpp testworld.h)
    target_link_libraries(${name} PRIVATE cr_core Qt6::Test)
    target_compile_definitions(${name} PRIVATE CR_SOURCE_DIR="${PROJECT_SOURCE_DIR}")
    add_test(NAME ${name} COMMAND ${name})
    set_tests_properties(${name} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endfunction()

cr_add_test(tst_countdown)
cr_add_test(tst_solar)
cr_add_test(tst_physics)
cr_add_test(tst_format)
cr_add_test(tst_schedule)
cr_add_test(tst_analysis)
cr_add_test(tst_worlddata)
